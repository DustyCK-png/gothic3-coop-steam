cmake_minimum_required(VERSION 3.24)
project(G3Steam C)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Steam message ABI requires Win32")
endif()
add_executable(G3Steam g3steam.c)
add_library(mock_steam_api SHARED mock_steam_api.c)
foreach(target G3Steam mock_steam_api)
  target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(${target} PRIVATE ws2_32)
endforeach()
